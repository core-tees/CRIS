cmake_minimum_required(VERSION 3.20)
project(CRIS LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(CRIS_PETSC "Build the CPU PETSc transport solver and label generator" OFF)
option(CRIS_CUDA "Build the CUDA/PETSc transport solver and label generator" OFF)
option(CRIS_BRATU_TRAINING "Build the LibTorch/MKL Bratu trainer" OFF)
find_package(OpenMP REQUIRED COMPONENTS CXX)

foreach(name column_neural column_chebyshev norne_reference)
  add_executable(${name} src/accuracy/${name}.cpp)
  target_link_libraries(${name} PRIVATE OpenMP::OpenMP_CXX)
endforeach()
add_executable(check_export src/transport/check_export.cpp)
target_link_libraries(check_export PRIVATE OpenMP::OpenMP_CXX)
if(CRIS_PETSC OR CRIS_CUDA)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(PETSC REQUIRED IMPORTED_TARGET PETSc)
  find_package(MPI REQUIRED COMPONENTS CXX)
endif()
if(CRIS_PETSC)
  add_executable(petsc_transport src/transport/cpu_main.cpp)
  target_link_libraries(petsc_transport PRIVATE PkgConfig::PETSC MPI::MPI_CXX OpenMP::OpenMP_CXX)
  add_executable(labels_cpu src/labels/labels_cpu.cpp)
  target_link_libraries(labels_cpu PRIVATE PkgConfig::PETSC MPI::MPI_CXX)
endif()
if(CRIS_CUDA)
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 80)
  endif()
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_executable(petsc_transport_gpu src/transport/gpu_main.cu)
  set_target_properties(petsc_transport_gpu PROPERTIES CUDA_STANDARD 17)
  target_compile_options(petsc_transport_gpu PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O3>)
  separate_arguments(CRIS_OPENMP_HOST_FLAGS NATIVE_COMMAND "${OpenMP_CXX_FLAGS}")
  foreach(flag IN LISTS CRIS_OPENMP_HOST_FLAGS)
    target_compile_options(petsc_transport_gpu PRIVATE "$<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=${flag}>")
  endforeach()
  target_link_libraries(petsc_transport_gpu PRIVATE PkgConfig::PETSC MPI::MPI_CXX CUDA::cublas OpenMP::OpenMP_CXX)
  add_executable(labels_gpu src/labels/labels.cu)
  set_target_properties(labels_gpu PROPERTIES CUDA_STANDARD 17)
  target_compile_options(labels_gpu PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--expt-relaxed-constexpr>)
  target_link_libraries(labels_gpu PRIVATE PkgConfig::PETSC MPI::MPI_CXX)
  add_executable(parity_coordinates src/labels/parity_coordinates.cu)
  set_target_properties(parity_coordinates PROPERTIES CUDA_STANDARD 17)
endif()
if(CRIS_BRATU_TRAINING)
  find_package(Torch REQUIRED)
  find_package(MKL CONFIG REQUIRED)
  add_executable(bratu_lm_trainer src/bratu/bratu_lm_trainer.cpp)
  target_include_directories(bratu_lm_trainer PRIVATE src)
  target_link_libraries(bratu_lm_trainer PRIVATE ${TORCH_LIBRARIES} MKL::MKL OpenMP::OpenMP_CXX)
  separate_arguments(CRIS_TORCH_FLAGS NATIVE_COMMAND "${TORCH_CXX_FLAGS}")
  target_compile_options(bratu_lm_trainer PRIVATE ${CRIS_TORCH_FLAGS})
endif()

enable_testing()
foreach(law quadratic intermediate singular)
  add_test(NAME transport_export_${law} COMMAND check_export
    ${CMAKE_CURRENT_SOURCE_DIR}/models/transport/shared976_${law}.model
    ${CMAKE_CURRENT_SOURCE_DIR}/models/transport/shared976_${law}.queries)
endforeach()
